export default function ArtistCard({ artist }: Props) {
  const [following, setFollowing] = useState(false);
  const [busy, setBusy] = useState(false);
  const [showToast, setShowToast] = useState<string | null>(null);

  const name = artist.full_name || artist.username || "Artist";
  const initials = name
    .split(" ")
    .map((w) => w[0])
    .join("")
    .slice(0, 2)
    .toUpperCase();

  const roleLabel =
    artist.role === "promoter"
      ? "Promoter"
      : artist.role === "venue"
      ? "Venue"
      : artist.role === "artist"
      ? "Artist"
      : "Profile";

  const toggleFollow = async (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    if (busy) return;
    setBusy(true);
    const prev = following;
    setFollowing(!prev);
    try {
      const res = await fetch("/api/follows/toggle", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ targetType: artist.role || "user", targetId: artist.id }),
      });
      const data = await res.json();
      if (typeof data.following === "boolean") setFollowing(data.following);
    } catch {
      setFollowing(prev);
    } finally {
      setBusy(false);
    }
  };

  const share = async (e: React.MouseEvent) => {
    e.preventDefault();
    e.stopPropagation();
    const url =
      window.location.origin +
      (artist.username ? "/artists/" + artist.username : "");
    try {
      if (navigator.share) {
        await navigator.share({ title: name, url });
      } else {
        await navigator.clipboard.writeText(url);
        setShowToast("Link copied");
        setTimeout(() => setShowToast(null), 1800);
      }
    } catch {
      /* cancelled */
    }
  };
